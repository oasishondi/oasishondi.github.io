      .setConfirmationMessage(S.thanks)
      .setCollectEmail(false)
      .setProgressBar(true)
      .setShowLinkToRespondAgain(false);
  let n = 0;
  S.blocks.forEach(function (blk) {
    if (blk[0] === 'part') {
      form.addPageBreakItem().setTitle(blk[2]);
      blk[3].forEach(function (q) {
        n++;
        addQuestion(form, q, 'Q' + n + '. ');
        // 제주 거주 확인 문항: '아니오'면 바로 제출
        if (q[2].indexOf('설문 종료') >= 0) form.addPageBreakItem().setTitle(blk[2]);
      });
    } else if (blk[0] === 'box') {
      form.addPageBreakItem().setTitle(blk[1])
          .setHelpText(blk[2].map(function (l) { return '• ' + l; }).join('\n'));
    } else if (blk[0] === 'tasks') {
      form.addPageBreakItem().setTitle('혼디 써 보기')
          .setHelpText('휴대폰으로 https://oasishondi.github.io 에 접속해 아래 과제를 순서대로 해 본 뒤 [다음]을 눌러 주세요.\n\n' +
            DATA.TASKS.map(function (t) { return t[0] + '  ' + t[1] + '\n    ' + t[2]; }).join('\n\n'));
    }
  });
  fixExitQuestion(form);
  Logger.log('■ ' + S.title + '\n  응답용 링크: ' + form.getPublishedUrl() + '\n  편집 링크: ' + form.getEditUrl());
}

function addQuestion(form, q, prefix) {
  const kind = q[0], title = prefix + q[1], hint = q[2];
  if (kind === 'choice') {
    let opts = q[3].slice(), other = false;
    if (opts[opts.length - 1].indexOf('기타') === 0) { opts.pop(); other = true; }
    const multi = hint.indexOf('중복') >= 0 || hint.indexOf('최대') >= 0;
    const item = multi ? form.addCheckboxItem() : form.addMultipleChoiceItem();
    item.setTitle(title).setChoiceValues(opts).showOtherOption(other).setRequired(true);
    if (hint) item.setHelpText(hint);
    const m = hint.match(/최대 (\d)개/);
    if (m) item.setValidation(FormApp.createCheckboxValidation().requireSelectAtMost(Number(m[1])).build());
  } else if (kind === 'grid') {
    const cols = q[3].length === 5 ? q[3].map(function (s, i) { return (i + 1) + ' ' + s; }) : q[3];
    form.addGridItem().setTitle(title).setHelpText(hint).setRows(q[4]).setColumns(cols).setRequired(true);
  } else if (kind === 'text') {
    form.addParagraphTextItem().setTitle(title).setHelpText(hint).setRequired(false);
  }
}

// '현재 제주도에 살고 있습니까?'에서 '아니오'를 고르면 설문 제출로 보냄
function fixExitQuestion(form) {
  const items = form.getItems(FormApp.ItemType.MULTIPLE_CHOICE);
  items.forEach(function (it) {
    const mc = it.asMultipleChoiceItem();
    if (mc.getTitle().indexOf('제주도에 살고') < 0) return;
    mc.setChoices([
      mc.createChoice('예', FormApp.PageNavigationType.CONTINUE),
      mc.createChoice('아니오', FormApp.PageNavigationType.SUBMIT)
    ]);
  });
}
